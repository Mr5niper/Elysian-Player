# Pinned build/runtime dependencies for Elysian Player.
#
# Every version is pinned with == so that what gets tested here is exactly
# what everyone else builds. BUILD_EXE.bat wipes and recreates .venv on every
# run, so nothing outside this file can leak into a build.
#
# Targets Python 3.13.12 on Windows (enforced by BUILD_EXE.bat).

# ---- Runtime dependencies ----
# pywebview     - the window. Draws the UI with WebView2, which is already part
#                 of Windows, so the interface costs about half a megabyte
#                 rather than shipping a browser engine.
# just_playback - audio on miniaudio. Replaced pygame.mixer, which could not
#                 seek without restarting the stream and reported position
#                 from a wall clock that drifted out of sync.
# miniaudio     - used directly to decode waveform peaks
# mutagen       - tags and duration for mp3, flac, wav and ogg
# pillow        - decodes embedded album art into a data URL for the frontend
pywebview==6.2.1
just_playback==0.1.8
miniaudio==1.71
mutagen==1.48.1
pillow==12.3.0

# ---- Build chain (used only by BUILD_EXE.bat) ----

altgraph==0.17.5
packaging==26.3
pefile==2024.8.26
pywin32-ctypes==0.2.3
